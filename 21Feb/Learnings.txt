Sieve Of Erastothenes -> O(N Log Log N)

//This can be made static so that we dont always need to compute isPrime because bits are always less 33
static boolean[] prime = new boolean[33];

static {
    Arrays.fill(prime, true);
    prime[0] = false;
    prime[1] = false;
    for(int i = 2; i * i < 33; i++){
        if(prime[i]){
            for(int j = i * i; j < 33; j += i){
                prime[j] = false;
            }
        }
    }
}
